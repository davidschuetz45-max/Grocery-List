        let householdCode = '';

        let aldiItems = {};

        function generateHouseholdCode() {

            return Math.random().toString(36).substring(2, 8).toUpperCase();

        }

        

        function loadData() {

            householdCode = localStorage.getItem('householdCode') || '';

            

            if (!householdCode) {

                document.getElementById('welcomeScreen').classList.remove('hidden');

                document.getElementById('listTab').classList.add('hidden');

                return;

            }

            

            document.getElementById('householdCodeDisplay').textContent = householdCode;

            document.getElementById('householdInfo').classList.remove('hidden');

            document.getElementById('welcomeScreen').classList.add('hidden');

            document.getElementById('listTab').classList.remove('hidden');

            

            // Load shared household data

            const savedHousehold = localStorage.getItem(`household:${householdCode}`);

            if (savedHousehold) {

                try {

                    const parsed = JSON.parse(savedHousehold);

                    aldiItems = parsed.aldiItems || {};

                    krogerItems = parsed.krogerItems || {};

                } catch (e) {

                    console.error('Error loading household data:', e);

                }

            }

            

            // Load personal data

            const savedRecipes = localStorage.getItem('recipes');

            const savedHistory = localStorage.getItem('historyLists');

            const savedMissing = localStorage.getItem('missingItems');

            

            if (savedRecipes) {

                try {

                    recipes = JSON.parse(savedRecipes);

                } catch (e) {

                    recipes = [];

                }

            }

            if (savedHistory) {

                try {

                    historyLists = JSON.parse(savedHistory);

                } catch (e) {

                    historyLists = [];

                }

            }

            if (savedMissing) {

                try {

                    missingItems = JSON.parse(savedMissing);

                } catch (e) {

                    missingItems = [];

                }

            }

            

            renderLists();

            renderRecipes();

            renderMissingItems();

            renderHistory();

        }

        

        function saveData() {

            if (!householdCode) return;

            

            const householdData = {

                aldiItems,

                krogerItems,

                lastUpdated: Date.now()

            };

            

            // Save household data (shared)

            localStorage.setItem(`household:${householdCode}`, JSON.stringify(householdData));

            

            // Save personal data

            localStorage.setItem('recipes', JSON.stringify(recipes));

            localStorage.setItem('historyLists', JSON.stringify(historyLists));

            localStorage.setItem('missingItems', JSON.stringify(missingItems));

        }

        

        function startNewList() {

            householdCode = generateHouseholdCode();

            localStorage.setItem('householdCode', householdCode);

            document.getElementById('welcomeScreen').classList.add('hidden');

            document.getElementById('listTab').classList.remove('hidden');

            document.getElementById('householdCodeDisplay').textContent = householdCode;

            document.getElementById('householdInfo').classList.remove('hidden');

            

            aldiItems = {};

            krogerItems = {};

            

            saveData();

            renderLists();

        }

        

        function joinHousehold() {

            const code = document.getElementById('joinCode').value.trim().toUpperCase();

            if (!code) {

                alert('Please enter a household code');

                return;

            }

            

            householdCode = code;

            localStorage.setItem('householdCode', householdCode);

            document.getElementById('welcomeScreen').classList.add('hidden');

            document.getElementById('listTab').classList.remove('hidden');

            document.getElementById('householdCodeDisplay').textContent = householdCode;

            document.getElementById('householdInfo').classList.remove('hidden');

            

            loadData();

        }

        function addItem() {

            const input = document.getElementById('newItem');

            const itemName = input.value.trim();

            if (!itemName) return;

            

            const defaultStore = shouldDefaultToKroger(itemName) ? 'kroger' : 'aldi';

            const category = categorizeItem(itemName, defaultStore);

            const item = {

                name: itemName,

                id: Date.now() + Math.random(),

                checked: false

            };

            

            if (defaultStore === 'aldi') {

                if (!aldiItems[category]) aldiItems[category] = [];

                aldiItems[category].push(item);

            } else {

                if (!krogerItems[category]) krogerItems[category] = [];

                krogerItems[category].push(item);

            }

            

            input.value = '';

            saveData();

            renderLists();

        }

        

        function toggleItem(store, category, itemId) {

            const items = store === 'aldi' ? aldiItems : krogerItems;

            const item = items[category].find(i => i.id === itemId);

            if (item) {

                item.checked = !item.checked;

                saveData();

                renderLists();

            }

        }

        

        function deleteItem(store, category, itemId) {

            const items = store === 'aldi' ? aldiItems : krogerItems;

            items[category] = items[category].filter(i => i.id !== itemId);

            if (items[category].length === 0) {

                delete items[category];

            }

            saveData();

            renderLists();

        }

        

        function moveToKroger(category, itemId) {

            const item = aldiItems[category].find(i => i.id === itemId);

            if (!item) return;

            

            aldiItems[category] = aldiItems[category].filter(i => i.id !== itemId);

            if (aldiItems[category].length === 0) {

                delete aldiItems[category];

            }

            

            const krogerCategory = KROGER_CATEGORIES[0];

            if (!krogerItems[krogerCategory]) krogerItems[krogerCategory] = [];

            krogerItems[krogerCategory].push({...item, id: Date.now() + Math.random()});

            

            saveData();

            renderLists();

        }
